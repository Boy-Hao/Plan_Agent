## 3. 回调入口

```python
@router.post("/api/bots/{channel}/callback/{bot_id}")
async def callback(channel: str, bot_id: str, request: Request):
    bot = await bots.get_or_404(bot_id)
    if not bot.enabled:
        raise BotDisabled()
    adapter = registry.get(channel, bot)
    raw = await request.body()
    if not await adapter.verify(RawRequest(request.headers, raw)):
        raise InvalidSignature()
    events = await adapter.parse(RawRequest(request.headers, raw))
    for ev in events:
        if not await dedupe.first(bot.id, ev.event_id):
            continue
        await queue.enqueue_inbound(bot, ev)
    return {"ok": True}
```

**要点**：先验签再解析；去重后入队；快速 200。

## 4. 入站 Worker

```python
async def handle_inbound(bot, ev: InboundMessage):
    identity = await identity.resolve(bot, ev.sender)
    if not identity.ok:
        await outbound.send_bind_prompt(bot, ev)
        return
    agent = await router.resolve(bot, ev)
    session = await conversations.get_or_create(bot, ev, identity, agent)
    async for event in runtime.chat(session, ev.text):
        await outbound.push(bot, ev.chat_id, event)
    await outbound.finalize(bot, ev.chat_id)
```

## 5. 出站渲染

```python
class Renderer:
    def __init__(self, adapter): self.adapter = adapter; self.caps = adapter.capabilities()

    async def push(self, chat_id, event):
        if event.type == "token":
            if self.caps.streaming and self.caps.edit_message:
                await self._throttled_update(chat_id, event)
            return
        if event.type == "tool_call_start":
            await self._status(chat_id, "正在调用工具…")
        if event.type == "msg_end":
            await self._final(chat_id, event)

    async def _final(self, chat_id, event):
        text = event.text
        if self.caps.cards and event.card:
            return await self.adapter.send_card(chat_id, event.card)
        return await self.adapter.send_text(chat_id, chunk(text, self.caps.max_text_len))
```

**降级**：卡片 → Markdown → 文本 → 摘要 + 链接。

```python
async def first(bot_id: str, event_id: str) -> bool:
    key = f"t:{tenant}:bot:dedupe:{bot_id}:{event_id}"
    return await redis.set(key, "1", nx=True, ex=86400)
```
